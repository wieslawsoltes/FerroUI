using System;

class Program
{
    #region Greeting
    static string Greeting(string name)
    {
        return $"Hello, {name}!";
    }
    #endregion

    #region Entry point
    static void Main()
    {
        foreach (var name in new[] { "FerroUI", "C#", "😀" })
        {
            Console.WriteLine(Greeting(name));
        }
    }
    #endregion
}
